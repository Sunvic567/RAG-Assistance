#use this SQL to create the memory table in your Postgres database
create table agent_memory (
    id uuid primary key default uuid_generate_v4(),
    session_id text not null,
    role text not null,              -- "user" or "assistant"
    message text,                    -- original message
    pinecone_id text,                -- optional: link to pinecone chunk
    created_at timestamptz default now()
);

create index agent_memory_session_idx on agent_memory(session_id);
